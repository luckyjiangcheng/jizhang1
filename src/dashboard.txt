<script>
    // --- State ---
    let allData = [];
    let currentPeriod = 'month'; // default
    let charts = {};

    // --- Config ---
    const CSV_PLACEHOLDER_KEY = "CSV_DATA_" + "PLACEHOLDER";
    
    // Default Sample Data (if placeholder not replaced)
    const SAMPLE_DATA = `Date,Time,Amount,Category,Item,Merchant
2023-10-01,08:00,15.00,餐饮,早餐,KFC
2023-10-02,12:30,35.00,餐饮,午餐,McDonalds

    let csvContent = `CSV_DATA_PLACEHOLDER`;

    // --- Init ---
    document.addEventListener('DOMContentLoaded', () => {
        try {
            // Load Data
            if (csvContent.includes(CSV_PLACEHOLDER_KEY)) {
                console.log("Using sample data");
                allData = parseCSV(SAMPLE_DATA);
            } else {
                allData = parseCSV(csvContent);
            }

            // Set Initial View
            setPeriod('month');

            // Resize Handler
            window.addEventListener('resize', () => {
                Object.values(charts).forEach(c => c && c.resize());
            });

        } catch (e) {
            showError(e);
        }
    });

    function showError(e) {
        console.error(e);
        const el = document.getElementById('debug-log');
        el.style.display = 'block';
        el.innerText = `Error: ${e.message}`;
    }

    // --- Data Parsing ---
    function parseCSV(csv) {
        const lines = csv.replace(/\r\n/g, '\n').replace(/\r/g, '\n').trim().split('\n');
        if (lines.length < 2) return [];

        const headers = lines[0].split(',').map(h => h.trim());
        
        return lines.slice(1).map(line => {
            const row = line.split(',');
            if (row.length < headers.length) return null; // Skip bad rows

            const item = {};
            headers.forEach((h, i) => {
                item[h] = row[i] ? row[i].trim() : '';
            });

            // Parse Numbers
            item.Amount = parseFloat(item.Amount) || 0;
            
            // Parse Date
            if (item.Date) {
                // Combine Date + Time if available
                const timeStr = item.Time ? item.Time : '00:00:00';
                item.dateObj = new Date(item.Date.replace(/-/g, '/') + ' ' + timeStr);
                // Fallback if invalid
                if (isNaN(item.dateObj.getTime())) {
                     item.dateObj = new Date(item.Date.replace(/-/g, '/'));
                }
            } else {
                item.dateObj = new Date(0);
            }
            return item;
        }).filter(item => item && !isNaN(item.dateObj.getTime()));
    }

    // --- Core Logic ---
    function setPeriod(period) {
        currentPeriod = period;
        
        // Update Tabs UI
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(`tab-${period}`).classList.add('active');

        // Calculate Date Ranges
        const now = new Date(); // "Today" base
        let start, end, prevStart, prevEnd;
        let granularity = 'day';

        // Reset hours to start of day for accurate comparisons
        const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate());
        
        switch (period) {
            case 'today':
                start = new Date(todayStart);
                end = new Date(todayStart);
                end.setHours(23, 59, 59, 999);
                
                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 1);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 1);
                granularity = 'hour';
                break;

            case 'week':
                // Current Week (Mon-Sun)
                const day = todayStart.getDay() || 7; // 1=Mon, 7=Sun
                start = new Date(todayStart);
                start.setDate(todayStart.getDate() - day + 1); // Mon
                end = new Date(todayStart); // Up to today? Or end of week? Let's say end of today for realistic "current week" tracking, or end of week. Usually end of week.
                end.setDate(start.getDate() + 6);
                end.setHours(23, 59, 59, 999);

                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 7);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 7);
                granularity = 'day';
                break;

            case 'month':
                start = new Date(todayStart.getFullYear(), todayStart.getMonth(), 1);
                end = new Date(todayStart.getFullYear(), todayStart.getMonth() + 1, 0, 23, 59, 59);
                
                prevStart = new Date(todayStart.getFullYear(), todayStart.getMonth() - 1, 1);
                prevEnd = new Date(todayStart.getFullYear(), todayStart.getMonth(), 0, 23, 59, 59);
                granularity = 'day';
                break;

            case 'year':
                start = new Date(todayStart.getFullYear(), 0, 1);
                end = new Date(todayStart.getFullYear(), 11, 31, 23, 59, 59);
                
                prevStart = new Date(todayStart.getFullYear() - 1, 0, 1);
                prevEnd = new Date(todayStart.getFullYear() - 1, 11, 31, 23, 59, 59);
                granularity = 'month';
                break;
        }

        // Update Label
        const fmtDate = (d) => d.toISOString().slice(0, 10);
        document.getElementById('current-range-label').innerText = period === 'today' ? fmtDate(start) : `${fmtDate(start)}\n~\n${fmtDate(end)}`;

        // Filter Data
        const currentData = allData.filter(d => d.dateObj >= start && d.dateObj <= end);
        const prevData = allData.filter(d => d.dateObj >= prevStart && d.dateObj <= prevEnd);

        renderMetrics(currentData, prevData);
        renderCharts(currentData, granularity, start, end);
        renderList(currentData);
    }

    function renderMetrics(curr, prev) {
        // Total
        const totalCurr = curr.reduce((sum, item) => sum + item.Amount, 0);
        const totalPrev = prev.reduce((sum, item) => sum + item.Amount, 0);
        
        // Count
        const countCurr = curr.length;
        const countPrev = prev.length;

        // Avg
        const avgCurr = countCurr ? totalCurr / countCurr : 0;
        const avgPrev = countPrev ? totalPrev / countPrev : 0;

        // Helper: Calculate Trend
        const calcTrend = (c, p) => {
            if (p === 0) return { val: c === 0 ? '0%' : '+∞%', type: 'neutral' };
            const pct = ((c - p) / p) * 100;
            const sign = pct > 0 ? '↑' : (pct < 0 ? '↓' : '');
            const type = pct > 0 ? 'trend-up' : (pct < 0 ? 'trend-down' : 'trend-neutral');
            return { val: `${sign} ${Math.abs(pct).toFixed(1)}%`, type };
        };

        const tTotal = calcTrend(totalCurr, totalPrev);
        const tAvg = calcTrend(avgCurr, avgPrev);
        const tCount = calcTrend(countCurr, countPrev);

        // Update DOM
        document.getElementById('val-total').innerText = '¥' + totalCurr.toLocaleString('zh-CN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
        document.getElementById('trend-total').innerHTML = `<span class="${tTotal.type}">${tTotal.val}</span> <span style="color:#9ca3af">环比</span>`;

        document.getElementById('val-avg').innerText = '¥' + avgCurr.toLocaleString('zh-CN', {minimumFractionDigits: 0, maximumFractionDigits: 2});
        document.getElementById('trend-avg').innerHTML = `<span class="${tAvg.type}">${tAvg.val}</span> <span style="color:#9ca3af">环比</span>`;

        document.getElementById('val-count').innerText = countCurr;
        document.getElementById('trend-count').innerHTML = `<span class="${tCount.type}">${tCount.val}</span> <span style="color:#9ca3af">环比</span>`;
    }

    function renderCharts(data, granularity, start, end) {
        const isDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
        const textColor = isDark ? '#aeaeb2' : '#8e8e93';
        const splitLineColor = isDark ? '#38383a' : '#f2f2f7';

        // 1. Pie Chart (By Category)
        const catMap = {};
        data.forEach(d => {
            const c = d.Category || '未分类';
            catMap[c] = (catMap[c] || 0) + d.Amount;
        });
        const pieData = Object.entries(catMap)
            .map(([name, value]) => ({ name, value }))
            .sort((a, b) => b.value - a.value);

        initChart('chart-pie', {
            color: ['#007aff', '#34c759', '#ff9500', '#ff3b30', '#af52de', '#ff2d55', '#5856d6'],
            tooltip: { 
                trigger: 'item', 
                backgroundColor: isDark ? '#1c1c1e' : '#ffffff',
                borderColor: isDark ? '#38383a' : '#e5e7eb',
                textStyle: { color: isDark ? '#ffffff' : '#1c1c1e' }
            },
            legend: { 
                bottom: 0, 
                left: 'center', 
                type: 'scroll',
                textStyle: { color: textColor }
            },
            series: [{
                type: 'pie',
                radius: ['50%', '75%'],
                center: ['50%', '45%'],
                avoidLabelOverlap: false,
                itemStyle: {
                    borderRadius: 10,
                    borderColor: isDark ? '#1c1c1e' : '#fff',
                    borderWidth: 2
                },
                label: { show: false },
                emphasis: { label: { show: false } },
                data: pieData.length ? pieData : [{name:'无数据', value:0}]
            }]
        });

        // 2. Trend Chart
        const trendMap = {};
        
        // Initialize Trend Map with 0s for complete timeline
        if (granularity === 'hour') {
            for(let i=0; i<24; i++) trendMap[i] = 0;
            data.forEach(d => {
                const h = d.dateObj.getHours();
                trendMap[h] = (trendMap[h] || 0) + d.Amount;
            });
        } else if (granularity === 'day') {
            // Fill days between start and end
            for (let d = new Date(start); d <= end; d.setDate(d.getDate() + 1)) {
                const k = d.toISOString().slice(0, 10);
                trendMap[k] = 0;
            }
            data.forEach(d => {
                const k = d.dateObj.toISOString().slice(0, 10);
                trendMap[k] = (trendMap[k] || 0) + d.Amount;
            });
        } else if (granularity === 'month') {
            for(let i=0; i<12; i++) trendMap[i] = 0;
            data.forEach(d => {
                const m = d.dateObj.getMonth();
                trendMap[m] = (trendMap[m] || 0) + d.Amount;
            });
        }

        const xData = Object.keys(trendMap);
        const yData = Object.values(trendMap);
        
        // Format X Axis Labels
        const xLabels = xData.map(k => {
            if (granularity === 'hour') return `${k}:00`;
            if (granularity === 'month') return `${parseInt(k)+1}月`;
            if (granularity === 'day') return k.slice(5); // MM-DD
            return k;
        });

        initChart('chart-trend', {
            tooltip: { 
                trigger: 'axis',
                backgroundColor: isDark ? '#1c1c1e' : '#ffffff',
                borderColor: isDark ? '#38383a' : '#e5e7eb',
                textStyle: { color: isDark ? '#ffffff' : '#1c1c1e' }
            },
            grid: { left: '2%', right: '2%', bottom: '3%', top: '10%', containLabel: true },
            xAxis: { 
                type: 'category', 
                data: xLabels,
                axisLine: { show: false },
                axisTick: { show: false },
                axisLabel: { color: textColor, fontSize: 11 }
            },
            yAxis: { 
                type: 'value',
                splitLine: { lineStyle: { color: splitLineColor } },
                axisLabel: { color: textColor, fontSize: 11 }
            },
            series: [{
                data: yData,
                type: 'line',
                smooth: true,
                showSymbol: false,
                lineStyle: { width: 3, color: '#007aff' },
                areaStyle: {
                    color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                        { offset: 0, color: 'rgba(0, 122, 255, 0.3)' },
                        { offset: 1, color: 'rgba(0, 122, 255, 0)' }
                    ])
                },
                itemStyle: { color: '#007aff' }
            }]
        });
    }

    function renderList(data) {
        const listEl = document.getElementById('transaction-list');
        listEl.innerHTML = '';

        // Sort by Time Desc (Date + Time)
        const sorted = [...data].sort((a, b) => b.dateObj - a.dateObj);

        if (sorted.length === 0) {
            listEl.innerHTML = '<li class="empty-state">本时段无交易记录</li>';
            return;
        }

        // Limit to 50 items for performance
        const displayList = sorted.slice(0, 50);

        displayList.forEach(item => {
            const li = document.createElement('li');
            li.className = 'transaction-item';
            
            // Format Time
            const timeStr = item.dateObj.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' });
            const dateStr = item.dateObj.toLocaleDateString('zh-CN', { month: '2-digit', day: '2-digit' });

            li.innerHTML = `
                <div class="tx-left">
                    <span class="tx-category">${item.Category || '未分类'} <span style="font-weight:normal; font-size:0.9em"> - ${item.Item || ''}</span></span>
                    <span class="tx-meta">${dateStr} ${timeStr} · ${item.Merchant || ''}</span>
                </div>
                <div class="tx-right">
                    <span class="tx-amount">¥${item.Amount.toFixed(2)}</span>
                </div>
            `;
            listEl.appendChild(li);
        });
    }

    function initChart(domId, option) {
        const dom = document.getElementById(domId);
        if (!dom) return;
        
        let chart = charts[domId];
        if (!chart) {
            chart = echarts.init(dom);
            charts[domId] = chart;
        }
        chart.setOption(option);
    }
