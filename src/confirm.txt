<script>
    let rawData = `JSON_DATA_PLACEHOLDER`;
    const placeholderToken = ["JSON", "DATA", "PLACEHOLDER"].join("_");
    const categories = ["餐饮美食","交通出行","购物消费","居家生活","休闲娱乐","人情往来","医疗健康","教育培训","金融保险","孝敬父母","其他"];

    const dateEl = document.getElementById("date");
    const timeEl = document.getElementById("time");
    const amountEl = document.getElementById("amount");
    const categoryEl = document.getElementById("category");
    const itemEl = document.getElementById("item");
    const merchantEl = document.getElementById("merchant");
    const errorEl = document.getElementById("error");
    const chipsEl = document.getElementById("chips");

    function endWith(value) {
        if (window.completion) {
            window.completion(value);
            return;
        }
        document.body.innerText = value || " ";
        setTimeout(() => {
            window.close();
        }, 0);
    }

    function selectCategory(name) {
        categoryEl.value = name;
        const chips = chipsEl.querySelectorAll(".chip");
        chips.forEach(chip => {
            chip.classList.toggle("active", chip.dataset.value === name);
        });
    }

    function initChips() {
        chipsEl.innerHTML = "";
        categories.forEach(name => {
            const chip = document.createElement("div");
            chip.className = "chip";
            chip.dataset.value = name;
            chip.innerText = name;
            chip.addEventListener("click", () => selectCategory(name));
            chipsEl.appendChild(chip);
        });
    }

    function parseJsonData() {
        if (rawData === placeholderToken) return {};
        try {
            return JSON.parse(rawData);
        } catch (_) {
            return {};
        }
    }

    function fillDefaults() {
        const now = new Date();
        const yyyy = now.getFullYear();
        const mm = String(now.getMonth() + 1).padStart(2, "0");
        const dd = String(now.getDate()).padStart(2, "0");
        const hh = String(now.getHours()).padStart(2, "0");
        const mi = String(now.getMinutes()).padStart(2, "0");
        const data = parseJsonData();

        dateEl.value = data.date || `${yyyy}-${mm}-${dd}`;
        timeEl.value = data.time || `${hh}:${mi}`;
        if (data.amount !== undefined && data.amount !== null && data.amount !== "") {
            amountEl.value = data.amount;
        }
        if (data.item) itemEl.value = data.item;
        if (data.merchant) merchantEl.value = data.merchant;
        if (data.category && categories.includes(data.category)) {
            selectCategory(data.category);
        }
    }

    function toCsvLine() {
        const date = dateEl.value || "";
        const time = timeEl.value || "";
        const amount = amountEl.value || "";
        const category = categoryEl.value || "";
        const item = (itemEl.value || "").replace(/,/g, " ");
        const merchant = (merchantEl.value || "").replace(/,/g, " ");
        return `${date},${time},${amount},${category},${item},${merchant}`;
    }

    document.getElementById("cancelBtn").addEventListener("click", () => {
        endWith("");
    });

    document.getElementById("form").addEventListener("submit", (e) => {
        e.preventDefault();
        errorEl.innerText = "";
        if (!amountEl.value || !categoryEl.value) {
            errorEl.innerText = "请填写金额并选择类目";
            return;
        }
        endWith(toCsvLine());
    });

    initChips();
    fillDefaults();
</script>
