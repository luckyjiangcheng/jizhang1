<script>
    let rawData = `JSON_DATA_PLACEHOLDER`;
    const placeholderToken = ["JSON", "DATA", "PLACEHOLDER"].join("_");
    const categories = [
        "🍔 餐饮美食", "🚗 交通出行", "🛍️ 购物消费", "🏠 居家生活", 
        "🎮 休闲娱乐", "🧧 人情往来", "🏥 医疗健康", "🎓 教育培训", 
        "🛡️ 金融保险", "👴 孝敬父母", "✨ 其他"
    ];

    function showErrorDialog(title, detail) {
        const existing = document.getElementById("runtimeErrorDialog");
        if (existing) existing.remove();
        const overlay = document.createElement("div");
        overlay.id = "runtimeErrorDialog";
        overlay.style.cssText = "position:fixed;inset:0;background:rgba(0,0,0,.45);z-index:9999;display:flex;align-items:center;justify-content:center;padding:20px;";
        const box = document.createElement("div");
        box.style.cssText = "width:100%;max-width:420px;background:var(--card);color:var(--text);border-radius:16px;padding:16px;box-shadow:0 12px 30px rgba(0,0,0,.25);";
        const t = document.createElement("div");
        t.style.cssText = "font-size:17px;font-weight:700;margin-bottom:8px;";
        t.textContent = title || "发生错误";
        const d = document.createElement("div");
        d.style.cssText = "font-size:13px;line-height:1.5;color:var(--text-sec);white-space:pre-wrap;word-break:break-word;max-height:45vh;overflow:auto;";
        d.textContent = detail || "未知错误";
        const actions = document.createElement("div");
        actions.style.cssText = "display:flex;gap:10px;margin-top:14px;";
        const copyBtn = document.createElement("button");
        copyBtn.type = "button";
        copyBtn.style.cssText = "flex:1;border:none;border-radius:10px;padding:10px 0;background:var(--bg);color:var(--text);font-size:14px;font-weight:600;";
        copyBtn.textContent = "复制错误";
        copyBtn.addEventListener("click", () => {
            const text = `[${t.textContent}] ${d.textContent}`;
            const copied = writeClipboardSync(text);
            if (!copied && navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(text).catch(() => {});
            }
        });
        const closeBtn = document.createElement("button");
        closeBtn.type = "button";
        closeBtn.style.cssText = "flex:1;border:none;border-radius:10px;padding:10px 0;background:var(--primary);color:#fff;font-size:14px;font-weight:600;";
        closeBtn.textContent = "我知道了";
        closeBtn.addEventListener("click", () => overlay.remove());
        actions.appendChild(copyBtn);
        actions.appendChild(closeBtn);
        box.appendChild(t);
        box.appendChild(d);
        box.appendChild(actions);
        overlay.appendChild(box);
        overlay.addEventListener("click", (e) => {
            if (e.target === overlay) overlay.remove();
        });
        document.body.appendChild(overlay);
    }

    const dateEl = document.getElementById("date");
    const timeEl = document.getElementById("time");
    const amountEl = document.getElementById("amount");
    const amountInlineEl = document.getElementById("amountInline");
    const categoryEl = document.getElementById("category");
    const itemEl = document.getElementById("item");
    const merchantEl = document.getElementById("merchant");
    const errorEl = document.getElementById("error");
    let clipboardLineCache = "";
    let autoSyncTimer = null;

    function endWith(value) {
        let lastError = "";
        const errText = (e) => (e && (e.message || String(e))) || "";
        const handlerNames = [];
        let handoffTriggered = false;
        const launchShortcutsNow = (url) => {
            try {
                const a = document.createElement("a");
                a.href = url;
                a.style.display = "none";
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                handoffTriggered = true;
            } catch (e) {
                lastError = errText(e);
            }
            try {
                window.location.assign(url);
                handoffTriggered = true;
            } catch (e) {
                lastError = errText(e);
            }
            try {
                window.location.href = url;
                handoffTriggered = true;
            } catch (e) {
                lastError = errText(e);
            }
        };
        try {
            if (typeof window.completion === "function") window.completion(value);
        } catch (e) {
            lastError = errText(e);
        }
        try {
            if (typeof completion === "function") completion(value);
        } catch (e) {
            lastError = errText(e);
        }
        try {
            const handlers = window.webkit && window.webkit.messageHandlers ? window.webkit.messageHandlers : null;
            if (handlers) {
                Object.keys(handlers).forEach((name) => handlerNames.push(name));
                const preferred = ["completion", "close", "done", "dismiss", "callback"];
                const queue = preferred.concat(handlerNames.filter((n) => preferred.indexOf(n) === -1));
                queue.forEach((name) => {
                    const h = handlers[name];
                    if (h && typeof h.postMessage === "function") {
                        try {
                            h.postMessage(value);
                        } catch (e) {
                            lastError = errText(e);
                        }
                    }
                });
            }
        } catch (e) {
            lastError = errText(e);
        }
        launchShortcutsNow("shortcuts://");
        launchShortcutsNow("shortcuts://x-callback-url/return");
        setTimeout(() => {
            launchShortcutsNow("shortcuts://");
        }, 80);
        setTimeout(() => {
            launchShortcutsNow("shortcuts://x-callback-url/return");
        }, 180);
        setTimeout(() => {
            try {
                window.close();
            } catch (e) {
                lastError = errText(e);
            }
        }, 320);
        setTimeout(() => {
            try {
                history.back();
            } catch (e) {
                lastError = errText(e);
            }
        }, 460);
        setTimeout(() => {
            const detail = [
                lastError || "当前环境拦截了关闭动作，请点右上角对勾完成。",
                `window.completion: ${typeof window.completion}`,
                `completion: ${typeof completion}`,
                `webkit handlers: ${handlerNames.length ? handlerNames.join(", ") : "none"}`,
                `shortcuts handoff: ${handoffTriggered ? "attempted" : "not-triggered"}`
            ].join("\n");
            showErrorDialog("页面未能自动关闭", detail);
        }, 1300);
    }

    function initCategoryOptions() {
        categoryEl.innerHTML = "";
        categories.forEach(name => {
            const opt = document.createElement("option");
            opt.value = name;
            opt.innerText = name;
            categoryEl.appendChild(opt);
        });
    }

    function parseJsonData() {
        if (rawData === placeholderToken) return {};
        try {
            return JSON.parse(rawData);
        } catch (_) {
            return {};
        }
    }

    function fillDefaults() {
        const now = new Date();
        const yyyy = now.getFullYear();
        const mm = String(now.getMonth() + 1).padStart(2, "0");
        const dd = String(now.getDate()).padStart(2, "0");
        const hh = String(now.getHours()).padStart(2, "0");
        const mi = String(now.getMinutes()).padStart(2, "0");
        const data = parseJsonData();

        dateEl.value = data.date || `${yyyy}-${mm}-${dd}`;
        timeEl.value = data.time || `${hh}:${mi}`;
        if (data.amount !== undefined && data.amount !== null && data.amount !== "") {
            amountEl.value = data.amount;
            amountInlineEl.value = data.amount;
        }
        if (data.item) itemEl.value = data.item;
        if (data.merchant) merchantEl.value = data.merchant;
        
        if (data.category) {
            const match = categories.find(c => c === data.category || c.includes(data.category));
            categoryEl.value = match || data.category;
        } else if (categories.length) {
            categoryEl.value = categories[0];
        }
    }

    function toCsvLine() {
        const date = dateEl.value || "";
        const time = timeEl.value || "";
        const amount = amountEl.value || "";
        const category = categoryEl.value || "";
        const item = (itemEl.value || "").replace(/,/g, " ");
        const merchant = (merchantEl.value || "").replace(/,/g, " ");
        return `${date},${time},${amount},${category},${item},${merchant},,`;
    }

    function writeClipboardSync(text) {
        const textarea = document.createElement("textarea");
        textarea.value = text;
        textarea.style.position = "fixed";
        textarea.style.opacity = "0";
        document.body.appendChild(textarea);
        textarea.focus();
        textarea.select();
        let copied = false;
        try {
            copied = document.execCommand("copy");
        } catch (_) {}
        document.body.removeChild(textarea);
        return copied;
    }

    function writeClipboardAsync(text) {
        if (!navigator.clipboard || !navigator.clipboard.writeText) {
            return;
        }
        navigator.clipboard.writeText(text).catch(() => {});
    }

    function syncClipboardLine(line, showFailure) {
        const copied = writeClipboardSync(line);
        if (copied) {
            clipboardLineCache = line;
            return true;
        }
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(line).then(() => {
                clipboardLineCache = line;
            }).catch((e) => {
                if (showFailure) {
                    showErrorDialog("保存失败", (e && (e.message || String(e))) || "无法写入剪贴板");
                }
            });
            return true;
        }
        if (showFailure) {
            showErrorDialog("保存失败", "当前环境不支持剪贴板写入");
        }
        return false;
    }

    function scheduleAutoClipboardSync() {
        if (autoSyncTimer) {
            clearTimeout(autoSyncTimer);
        }
        autoSyncTimer = setTimeout(() => {
            const line = toCsvLine();
            if (line === clipboardLineCache) {
                return;
            }
            syncClipboardLine(line, false);
        }, 220);
    }

    document.getElementById("cancelBtn").addEventListener("click", () => {
        endWith("");
    });

    document.getElementById("form").addEventListener("submit", (e) => {
        e.preventDefault();
        errorEl.innerText = "";
        if (!amountEl.value || !categoryEl.value) {
            errorEl.innerText = "请填写金额并选择类目";
            return;
        }
        const line = toCsvLine();
        if (!syncClipboardLine(line, true)) {
            return;
        }
        endWith(line);
    });

    window.addEventListener("error", (event) => {
        const msg = event && event.message ? event.message : "脚本执行错误";
        showErrorDialog("运行错误", msg);
    });
    window.addEventListener("unhandledrejection", (event) => {
        const reason = event && event.reason ? event.reason : "Promise 未处理异常";
        const msg = typeof reason === "string" ? reason : (reason.message || JSON.stringify(reason));
        showErrorDialog("异步错误", msg);
    });

    amountEl.addEventListener("input", () => {
        if (amountInlineEl.value !== amountEl.value) amountInlineEl.value = amountEl.value;
        scheduleAutoClipboardSync();
    });
    amountInlineEl.addEventListener("input", () => {
        if (amountEl.value !== amountInlineEl.value) amountEl.value = amountInlineEl.value;
        scheduleAutoClipboardSync();
    });
    [dateEl, timeEl, categoryEl, itemEl, merchantEl].forEach((el) => {
        el.addEventListener("input", scheduleAutoClipboardSync);
        el.addEventListener("change", scheduleAutoClipboardSync);
    });

    initCategoryOptions();
    fillDefaults();
    if (!amountInlineEl.value && amountEl.value) amountInlineEl.value = amountEl.value;
    scheduleAutoClipboardSync();
</script>
