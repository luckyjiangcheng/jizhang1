    // --- Data Parsing ---
    function parseCSV(csv) {
        const lines = csv.replace(/\r\n/g, '\n').replace(/\r/g, '\n').trim().split('\n');
        if (lines.length < 2) return [];

        const headers = lines[0].split(',').map(h => h.trim());
        
        return lines.slice(1).map(line => {
            const row = line.split(',');
            if (row.length < headers.length) return null; // Skip bad rows

            const item = {};
            headers.forEach((h, i) => {
                item[h] = row[i] ? row[i].trim() : '';
            });

            // Parse Numbers
            item.Amount = parseFloat(item.Amount) || 0;
            
            // Parse Date
            if (item.Date) {
                // Combine Date + Time if available
                const timeStr = item.Time ? item.Time : '00:00:00';
                item.dateObj = new Date(item.Date.replace(/-/g, '/') + ' ' + timeStr);
                // Fallback if invalid
                if (isNaN(item.dateObj.getTime())) {
                     item.dateObj = new Date(item.Date.replace(/-/g, '/'));
                }
            } else {
                item.dateObj = new Date(0);
            }
            return item;
        }).filter(item => item && !isNaN(item.dateObj.getTime()));
    }

    // --- Helper Functions ---
    const toLocalDateString = (d) => {
        const year = d.getFullYear();
        const month = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
    };

    function setPeriod(period) {
        currentPeriod = period;
        
        // Update Tabs UI
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(`tab-${period}`).classList.add('active');

        // Calculate Date Ranges
        const now = new Date(); // "Today" base
        let start, end, prevStart, prevEnd;
        let granularity = 'day';

        // Reset hours to start of day for accurate comparisons
        const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate());
        
        switch (period) {
            case 'today':
                start = new Date(todayStart);
                end = new Date(todayStart);
                end.setHours(23, 59, 59, 999);
                
                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 1);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 1);
                granularity = 'hour';
                break;

            case 'week':
                // Current Week (Mon-Sun)
                const day = todayStart.getDay() || 7; // 1=Mon, 7=Sun
                start = new Date(todayStart);
                start.setDate(todayStart.getDate() - day + 1); // Mon
                end = new Date(todayStart);
                end.setDate(start.getDate() + 6);
                end.setHours(23, 59, 59, 999);

                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 7);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 7);
                granularity = 'day';
                break;

            case 'month':
                start = new Date(todayStart.getFullYear(), todayStart.getMonth(), 1);
                end = new Date(todayStart.getFullYear(), todayStart.getMonth() + 1, 0, 23, 59, 59);
                
                prevStart = new Date(todayStart.getFullYear(), todayStart.getMonth() - 1, 1);
                prevEnd = new Date(todayStart.getFullYear(), todayStart.getMonth(), 0, 23, 59, 59);
                granularity = 'day';
                break;

            case 'year':
                start = new Date(todayStart.getFullYear(), 0, 1);
                end = new Date(todayStart.getFullYear(), 11, 31, 23, 59, 59);
                
                prevStart = new Date(todayStart.getFullYear() - 1, 0, 1);
                prevEnd = new Date(todayStart.getFullYear() - 1, 11, 31, 23, 59, 59);
                granularity = 'month';
                break;
        }

        // Update Label
        document.getElementById('current-range-label').innerText = period === 'today' ? toLocalDateString(start) : `${toLocalDateString(start)}\n~\n${toLocalDateString(end)}`;

        // Filter Data
        const currentData = allData.filter(d => d.dateObj >= start && d.dateObj <= end);
        const prevData = allData.filter(d => d.dateObj >= prevStart && d.dateObj <= prevEnd);

        renderMetrics(currentData, prevData);
        renderCharts(currentData, granularity, start, end);
        renderList(currentData);
    }

    function renderMetrics(curr, prev) {
        // Calculate Income vs Expense
        // For simplicity, we assume positive Amount is Expense, 
        // unless Category is '收入' (case insensitive or match mapping)
        
        const calcTotals = (data) => {
            let inc = 0, exp = 0;
            data.forEach(item => {
                if (item.Category === '收入' || item.Amount < 0) {
                    inc += Math.abs(item.Amount);
                } else {
                    exp += item.Amount;
                }
            });
            return { inc, exp, bal: inc - exp };
        };

        const current = calcTotals(curr);
        
        // Update DOM
        const fmt = (v) => '¥' + v.toLocaleString('zh-CN', {minimumFractionDigits: 2, maximumFractionDigits: 2});
        
        document.getElementById('val-income').innerText = fmt(current.inc);
        document.getElementById('val-expense').innerText = fmt(current.exp);
        document.getElementById('val-balance').innerText = fmt(current.bal);
        
        // Color balance
        const balEl = document.getElementById('val-balance');
        balEl.style.color = current.bal >= 0 ? 'var(--income)' : 'var(--expense)';
    }

    function renderCharts(data, granularity, start, end) {
        const isDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
        const textColor = isDark ? '#aeaeb2' : '#8e8e93';
        const splitLineColor = isDark ? '#38383a' : '#f2f2f7';

        // Filter out income for spending charts
        const expenseData = data.filter(d => d.Category !== '收入' && d.Amount > 0);

        // 1. Pie Chart (By Category)
        const catMap = {};
        expenseData.forEach(d => {
            const c = d.Category || '未分类';
            catMap[c] = (catMap[c] || 0) + d.Amount;
        });
        const pieData = Object.entries(catMap)
            .map(([name, value]) => ({ name, value }))
            .sort((a, b) => b.value - a.value);

        initChart('chart-pie', {
            color: ['#007aff', '#34c759', '#ff9500', '#ff3b30', '#af52de', '#ff2d55', '#5856d6'],
            tooltip: { 
                trigger: 'item', 
                formatter: '{b}: ¥{c} ({d}%)',
                backgroundColor: isDark ? '#1c1c1e' : '#ffffff',
                borderColor: isDark ? '#38383a' : '#e5e7eb',
                textStyle: { color: isDark ? '#ffffff' : '#1c1c1e' }
            },
            legend: { 
                bottom: 0, 
                left: 'center', 
                type: 'scroll',
                textStyle: { color: textColor }
            },
            series: [{
                type: 'pie',
                radius: ['40%', '70%'],
                center: ['50%', '45%'],
                avoidLabelOverlap: false,
                itemStyle: {
                    borderRadius: 8,
                    borderColor: isDark ? '#1c1c1e' : '#fff',
                    borderWidth: 2
                },
                label: { show: false },
                data: pieData.length ? pieData : [{name:'无支出', value:0}]
            }]
        });

        // 2. Trend Chart
        const trendMap = {};
        
        // Initialize Trend Map with 0s for complete timeline
        if (granularity === 'hour') {
            for(let i=0; i<24; i++) trendMap[i] = 0;
            expenseData.forEach(d => {
                const h = d.dateObj.getHours();
                trendMap[h] = (trendMap[h] || 0) + d.Amount;
            });
        } else if (granularity === 'day') {
            for (let d = new Date(start); d <= end; d.setDate(d.getDate() + 1)) {
                const k = toLocalDateString(d);
                trendMap[k] = 0;
            }
            expenseData.forEach(d => {
                const k = toLocalDateString(d.dateObj);
                trendMap[k] = (trendMap[k] || 0) + d.Amount;
            });
        } else if (granularity === 'month') {
            for(let i=0; i<12; i++) trendMap[i] = 0;
            expenseData.forEach(d => {
                const m = d.dateObj.getMonth();
                trendMap[m] = (trendMap[m] || 0) + d.Amount;
            });
        }

        const xData = Object.keys(trendMap);
        const yData = Object.values(trendMap);
        const xLabels = xData.map(k => {
            if (granularity === 'hour') return `${k}:00`;
            if (granularity === 'month') return `${parseInt(k)+1}月`;
            if (granularity === 'day') return k.slice(5);
            return k;
        });

        initChart('chart-trend', {
            tooltip: { 
                trigger: 'axis',
                backgroundColor: isDark ? '#1c1c1e' : '#ffffff',
                borderColor: isDark ? '#38383a' : '#e5e7eb',
                textStyle: { color: isDark ? '#ffffff' : '#1c1c1e' }
            },
            grid: { left: '4%', right: '4%', bottom: '10%', top: '10%', containLabel: true },
            xAxis: { 
                type: 'category', 
                data: xLabels,
                axisLine: { show: false },
                axisTick: { show: false },
                axisLabel: { color: textColor, fontSize: 10 }
            },
            yAxis: { 
                type: 'value',
                splitLine: { lineStyle: { color: splitLineColor, type: 'dashed' } },
                axisLabel: { color: textColor, fontSize: 10 }
            },
            series: [{
                data: yData,
                type: 'line',
                smooth: true,
                showSymbol: false,
                lineStyle: { width: 3, color: '#007aff' },
                areaStyle: {
                    color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                        { offset: 0, color: 'rgba(0, 122, 255, 0.2)' },
                        { offset: 1, color: 'rgba(0, 122, 255, 0)' }
                    ])
                },
                itemStyle: { color: '#007aff' }
            }]
        });
    }

    function renderList(data) {
        const listEl = document.getElementById('transaction-list');
        listEl.innerHTML = '';

        const sorted = [...data].sort((a, b) => b.dateObj - a.dateObj);
        if (sorted.length === 0) {
            listEl.innerHTML = '<li class="empty-state">本时段无交易记录</li>';
            return;
        }

        sorted.slice(0, 50).forEach(item => {
            const isIncome = item.Category === '收入' || item.Amount < 0;
            const emoji = categoryEmojis[item.Category] || "✨";
            const amountClass = isIncome ? 'amount-income' : 'amount-expense';
            const amountPrefix = isIncome ? '+' : '-';
            
            const timeStr = item.dateObj.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' });
            const dateStr = item.dateObj.toLocaleDateString('zh-CN', { month: '2-digit', day: '2-digit' });
            
            const metaParts = [dateStr, timeStr];
            if (item.Account) metaParts.push(item.Account);
            if (item.Item || item.Merchant) metaParts.push(item.Item || item.Merchant);
            
            const metaStr = metaParts.join(' · ');
            const notesHtml = item.Notes ? `<div class="tx-notes">${item.Notes}</div>` : '';

            const li = document.createElement('li');
            li.className = 'transaction-item';
            li.innerHTML = `
                <div class="tx-left">
                    <div class="tx-icon">${emoji}</div>
                    <div class="tx-info">
                        <span class="tx-category">${item.Category || '未分类'}</span>
                        <span class="tx-meta">${metaStr}</span>
                        ${notesHtml}
                    </div>
                </div>
                <div class="tx-right">
                    <span class="tx-amount ${amountClass}">${amountPrefix}¥${Math.abs(item.Amount).toFixed(2)}</span>
                </div>
            `;
            listEl.appendChild(li);
        });
    }

    function initChart(domId, option) {
        if (typeof echarts === 'undefined') {
            console.warn('ECharts not loaded');
            return;
        }
        const dom = document.getElementById(domId);
        if (!dom) return;
        
        let chart = charts[domId];
        if (!chart) {
            chart = echarts.init(dom);
            charts[domId] = chart;
        }
        chart.setOption(option);
    }
