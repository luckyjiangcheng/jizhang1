<script>
    // --- Global State ---
    let rawData = [];
    let filteredData = [];
    let charts = {};

    // --- Configuration ---
    // Do NOT change this placeholder pattern, it is used by external tools
    const CSV_PLACEHOLDER_KEY = "CSV_DATA_" + "PLACEHOLDER";
    
    // Sample Data for Preview
    const SAMPLE_DATA = `Date,Time,Amount,Category,Item,Merchant
2023-10-01,08:00,15.00,Dining,Breakfast,KFC
2023-10-02,12:30,35.00,Dining,Lunch,McDonalds
2023-10-03,18:00,120.00,Shopping,Shoes,Nike
2023-10-05,09:00,50.00,Transport,Taxi,Uber
2023-10-08,19:00,200.00,Entertainment,Movie,Cinema
2023-10-12,10:00,80.00,Grocery,Fruits,Supermarket
2023-10-15,14:00,45.00,Dining,Coffee,Starbucks
2023-11-01,09:00,60.00,Grocery,Weekly Shop,Walmart
2023-11-05,19:30,100.00,Dining,Dinner,Italian Place
2023-12-10,15:00,300.00,Shopping,Gifts,Mall`;

    let csvContent = `CSV_DATA_PLACEHOLDER`;

    // --- Error Handling ---
    window.onerror = function(message, source, lineno, colno, error) {
        const debugEl = document.getElementById('debug-log');
        debugEl.style.display = 'block';
        debugEl.innerText += `[Error] ${message} at line ${lineno}:${colno}\n`;
        return false;
    };

    // --- Initialization ---
    document.addEventListener('DOMContentLoaded', () => {
        try {
            // 1. Load Data
            if (csvContent.includes(CSV_PLACEHOLDER_KEY)) {
                console.log("Using sample data (Preview Mode)");
                rawData = parseCSV(SAMPLE_DATA);
                document.querySelector('header h1').innerHTML += ' <span style="font-size:0.6em; color:gray; font-weight:normal">(Preview)</span>';
            } else {
                rawData = parseCSV(csvContent);
            }

            // 2. Initialize Filters
            initFilters();

            // 3. Initial Render
            applyFilters();

            // 4. Resize Listener
            window.addEventListener('resize', () => {
                Object.values(charts).forEach(chart => chart && chart.resize());
            });

        } catch (e) {
            console.error(e);
            document.getElementById('debug-log').style.display = 'block';
            document.getElementById('debug-log').innerText = `Init Error: ${e.message}`;
        }
    });

    // --- Core Logic ---

    function parseCSV(csv) {
        const lines = csv.replace(/\r\n/g, '\n').replace(/\r/g, '\n').trim().split('\n');
        const nonEmptyLines = lines.filter(l => l.trim().length > 0);
        if (nonEmptyLines.length < 2) return [];

        const headers = nonEmptyLines[0].split(',').map(h => h.trim());
        
        return nonEmptyLines.slice(1).map(line => {
            // Handle simplistic CSV parsing (no quotes support for now as per previous logic)
            const row = line.split(',');
            const item = {};
            headers.forEach((h, i) => {
                item[h] = row[i] ? row[i].trim() : '';
            });
            
            // Type conversion
            item.Amount = parseFloat(item.Amount);
            if (isNaN(item.Amount)) item.Amount = 0;
            
            // Date parsing helpers
            if (item.Date) {
                item.dateObj = new Date(item.Date + (item.Time ? 'T' + item.Time : ''));
            } else {
                item.dateObj = new Date(0); // Invalid date fallback
            }

            return item;
        }).filter(item => item.Date); // Filter out rows without date
    }

    function initFilters() {
        if (rawData.length === 0) return;

        // Sort by date
        rawData.sort((a, b) => a.dateObj - b.dateObj);

        const startInput = document.getElementById('startDate');
        const endInput = document.getElementById('endDate');
        const granularitySelect = document.getElementById('granularity');

        // Set default range (Full range of data)
        if (rawData.length > 0) {
            startInput.value = rawData[0].Date;
            endInput.value = rawData[rawData.length - 1].Date;
        }

        // Event Listeners
        startInput.addEventListener('change', applyFilters);
        endInput.addEventListener('change', applyFilters);
        granularitySelect.addEventListener('change', () => {
            // Only re-render charts if granularity changes, but simplest to just re-apply all
            applyFilters();
        });
    }

    function applyFilters() {
        const startStr = document.getElementById('startDate').value;
        const endStr = document.getElementById('endDate').value;
        const startDate = startStr ? new Date(startStr) : new Date(-8640000000000000);
        const endDate = endStr ? new Date(endStr) : new Date(8640000000000000);
        
        // Adjust end date to include the full day (if it's just YYYY-MM-DD)
        endDate.setHours(23, 59, 59, 999);

        filteredData = rawData.filter(item => {
            return item.dateObj >= startDate && item.dateObj <= endDate;
        });

        renderAll();
    }

    // --- UI Logic ---

    function switchTab(tabId) {
        // Update Buttons
        document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active'));
        event.target.classList.add('active');

        // Update Content
        document.querySelectorAll('.tab-content').forEach(content => content.classList.remove('active'));
        document.getElementById('tab-' + tabId).classList.add('active');

        // Resize charts in the new tab (ECharts sometimes needs this when unhidden)
        setTimeout(() => {
            Object.values(charts).forEach(chart => chart && chart.resize());
        }, 50);
    }

    function renderAll() {
        renderOverview();
        renderAnalysis();
        renderDetails();
    }

    function renderOverview() {
        // 1. Summaries
        const total = filteredData.reduce((sum, item) => sum + item.Amount, 0);
        const count = filteredData.length;
        const avg = count > 0 ? total / count : 0;

        document.getElementById('stat-total').innerText = total.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2});
        document.getElementById('stat-avg').innerText = avg.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2});
        document.getElementById('stat-count').innerText = count;

        // YoY Comparison Logic
        const startStr = document.getElementById('startDate').value;
        const endStr = document.getElementById('endDate').value;
        if (startStr && endStr) {
            const startDate = new Date(startStr);
            const endDate = new Date(endStr);
            
            // Shift back 1 year
            const prevStartDate = new Date(startDate);
            prevStartDate.setFullYear(startDate.getFullYear() - 1);
            
            const prevEndDate = new Date(endDate);
            prevEndDate.setFullYear(endDate.getFullYear() - 1);
            
            const prevTotal = rawData.reduce((sum, item) => {
                if (item.dateObj >= prevStartDate && item.dateObj <= prevEndDate) {
                    return sum + item.Amount;
                }
                return sum;
            }, 0);
            
            let diffHtml = '';
            if (prevTotal === 0) {
                diffHtml = '<small style="color:gray">No data last year</small>';
            } else {
                const percent = ((total - prevTotal) / prevTotal) * 100;
                const color = percent > 0 ? 'red' : 'green'; // Red = Increased spending (bad?), Green = Decreased
                const arrow = percent > 0 ? '↑' : '↓';
                diffHtml = `<small style="color:${color}">${arrow} ${Math.abs(percent).toFixed(1)}% YoY</small>`;
            }
            
            // Append to Total Stat
            document.getElementById('stat-total').innerHTML += `<br>${diffHtml}`;
        }

        // 2. Pie Chart (Category)
        const categoryMap = {};
        filteredData.forEach(item => {
            const cat = item.Category || 'Uncategorized';
            categoryMap[cat] = (categoryMap[cat] || 0) + item.Amount;
        });

        const pieData = Object.entries(categoryMap)
            .map(([name, value]) => ({ name, value }))
            .sort((a, b) => b.value - a.value);

        initChart('chart-pie', {
            title: { text: 'Expenses by Category', left: 'center' },
            tooltip: { trigger: 'item', formatter: '{b}: {c} ({d}%)' },
            legend: { orient: 'horizontal', bottom: 'bottom' },
            series: [{
                type: 'pie',
                radius: ['40%', '70%'],
                avoidLabelOverlap: false,
                itemStyle: { borderRadius: 10, borderColor: '#fff', borderWidth: 2 },
                label: { show: false, position: 'center' },
                emphasis: {
                    label: { show: true, fontSize: 20, fontWeight: 'bold' }
                },
                data: pieData
            }]
        });
    }

    function renderAnalysis() {
        const granularity = document.getElementById('granularity').value;
        
        // 1. Trend Chart (Bar)
        const aggregated = {};
        
        filteredData.forEach(item => {
            let key;
            const d = item.dateObj;
            if (granularity === 'day') {
                key = item.Date; // YYYY-MM-DD
            } else if (granularity === 'month') {
                key = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
            } else if (granularity === 'year') {
                key = `${d.getFullYear()}`;
            } else if (granularity === 'week') {
                // Simple Week calculation (ISO week would be better but this is dependency-free)
                const firstDayOfYear = new Date(d.getFullYear(), 0, 1);
                const pastDays = (d - firstDayOfYear) / 86400000;
                const weekNum = Math.ceil((pastDays + firstDayOfYear.getDay() + 1) / 7);
                key = `${d.getFullYear()}-W${weekNum}`;
            }
            aggregated[key] = (aggregated[key] || 0) + item.Amount;
        });

        const sortedKeys = Object.keys(aggregated).sort();
        const trendData = sortedKeys.map(k => aggregated[k]);

        initChart('chart-trend', {
            title: { text: 'Expense Trend', left: 'center' },
            tooltip: { trigger: 'axis' },
            xAxis: { type: 'category', data: sortedKeys },
            yAxis: { type: 'value' },
            series: [{
                data: trendData,
                type: 'bar',
                color: '#3b82f6',
                backgroundStyle: { color: 'rgba(180, 180, 180, 0.2)' }
            }]
        });

        // 2. Scatter Plot (Date vs Amount)
        // Helps identify outliers or spending habits by time
        const scatterData = filteredData.map(item => {
            return [
                item.Date, // X: Date
                item.Amount, // Y: Amount
                item.Category, // For tooltip
                item.Item
            ];
        });

        initChart('chart-scatter', {
            title: { text: 'Individual Transactions', left: 'center' },
            tooltip: {
                trigger: 'item',
                formatter: (params) => {
                    return `${params.data[0]}<br/>${params.data[2]} - ${params.data[3]}<br/><b>${params.data[1]}</b>`;
                }
            },
            xAxis: { type: 'time', splitLine: { show: false } },
            yAxis: { type: 'value', splitLine: { show: true } },
            series: [{
                symbolSize: 10,
                data: scatterData,
                type: 'scatter',
                itemStyle: { color: '#ef4444' }
            }]
        });
    }

    function renderDetails() {
        const tbody = document.querySelector('#details-table tbody');
        tbody.innerHTML = '';

        // Reverse chronological order for table
        const sortedForTable = [...filteredData].sort((a, b) => b.dateObj - a.dateObj);

        if (sortedForTable.length === 0) {
            tbody.innerHTML = '<tr><td colspan="6" style="text-align:center">No data found</td></tr>';
            return;
        }

        sortedForTable.forEach(item => {
            const tr = document.createElement('tr');
            tr.innerHTML = `
                <td>${item.Date}</td>
                <td>${item.Time || '-'}</td>
                <td>${item.Category || '-'}</td>
                <td>${item.Item || '-'}</td>
                <td>${item.Merchant || '-'}</td>
                <td style="text-align:right; font-weight:bold">${item.Amount.toFixed(2)}</td>
            `;
            tbody.appendChild(tr);
        });
    }

    function initChart(domId, option) {
        const dom = document.getElementById(domId);
        if (!dom) return;

        if (charts[domId]) {
            charts[domId].dispose();
        }
        
        charts[domId] = echarts.init(dom);
        charts[domId].setOption(option);
    }

</script>
