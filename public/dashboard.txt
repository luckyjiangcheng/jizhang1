    // --- Data Parsing ---
    function parseCSV(csv) {
        if (!csv || csv.length < 10) return [];
        const lines = csv.split(/\r?\n/);
        if (lines.length < 2) return [];

        const headers = lines[0].split(',').map(h => h.trim());
        const result = [];
        
        for (let i = lines.length - 1; i >= 1; i--) {
            const line = lines[i].trim();
            if (!line) continue;
            
            const row = line.split(',');
            if (row.length < headers.length) continue;

            const item = {};
            for (let j = 0; j < headers.length; j++) {
                let value = row[j] ? row[j].trim() : '';
                // 移除可能存在的引号
                value = value.replace(/^"|"$/g, '');
                item[headers[j]] = value;
            }

            // Parse Numbers
            item.Amount = parseFloat(item.Amount) || 0;
            
            // Parse Date
            if (item.Date) {
                let timeStr = item.Time || '00:00:00';
                // 处理非标准时间格式
                if (timeStr.length === 2) {
                    // 只有小时，补充分钟
                    timeStr = timeStr + ':00';
                } else if (timeStr.length === 4) {
                    // HHMM格式，转换为HH:MM
                    timeStr = timeStr.substring(0, 2) + ':' + timeStr.substring(2);
                }
                // Using a faster date parsing approach: YYYY/MM/DD HH:mm:ss
                item.dateObj = new Date(item.Date.replace(/-/g, '/') + ' ' + timeStr);
                if (isNaN(item.dateObj.getTime())) continue;
            } else {
                continue;
            }
            result.push(item);
        }
        return result;
    }

    // --- Helper Functions ---
    const toLocalDateString = (d) => {
        const year = d.getFullYear();
        const month = String(d.getMonth() + 1).padStart(2, '0');
        const day = String(d.getDate()).padStart(2, '0');
        return `${year}-${month}-${day}`;
    };

    function buildPeriodRange(period, todayStart) {
        let start, end, prevStart, prevEnd;
        let granularity = 'day';
        switch (period) {
            case 'today':
                start = new Date(todayStart);
                end = new Date(todayStart);
                end.setHours(23, 59, 59, 999);
                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 1);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 1);
                granularity = 'hour';
                break;
            case 'week':
                const day = todayStart.getDay() || 7;
                start = new Date(todayStart);
                start.setDate(todayStart.getDate() - day + 1);
                end = new Date(todayStart);
                end.setDate(start.getDate() + 6);
                end.setHours(23, 59, 59, 999);
                prevStart = new Date(start);
                prevStart.setDate(start.getDate() - 7);
                prevEnd = new Date(end);
                prevEnd.setDate(end.getDate() - 7);
                granularity = 'day';
                break;
            case 'month':
                start = new Date(todayStart.getFullYear(), todayStart.getMonth(), 1);
                end = new Date(todayStart.getFullYear(), todayStart.getMonth() + 1, 0, 23, 59, 59);
                prevStart = new Date(todayStart.getFullYear(), todayStart.getMonth() - 1, 1);
                prevEnd = new Date(todayStart.getFullYear(), todayStart.getMonth(), 0, 23, 59, 59);
                granularity = 'day';
                break;
            case 'year':
                start = new Date(todayStart.getFullYear(), 0, 1);
                end = new Date(todayStart.getFullYear(), 11, 31, 23, 59, 59);
                prevStart = new Date(todayStart.getFullYear() - 1, 0, 1);
                prevEnd = new Date(todayStart.getFullYear() - 1, 11, 31, 23, 59, 59);
                granularity = 'month';
                break;
        }
        return { start, end, prevStart, prevEnd, granularity };
    }

    function expenseStats(data, start, end) {
        const items = data.filter(d => d.dateObj >= start && d.dateObj <= end && d.Category !== '收入' && d.Amount > 0);
        return {
            amount: items.reduce((sum, item) => sum + item.Amount, 0),
            count: items.length
        };
    }

    function changeText(label, current, previous) {
        if (previous === 0) {
            if (current === 0) return `${label} 0.00%`;
            return `${label} 新增`;
        }
        const ratio = ((current - previous) / previous) * 100;
        const sign = ratio >= 0 ? '+' : '';
        return `${label} ${sign}${ratio.toFixed(2)}%`;
    }

    function applyChangeEl(id, label, current, previous) {
        const el = document.getElementById(id);
        if (!el) return;
        el.innerText = changeText(label, current, previous);
        const delta = current - previous;
        el.classList.remove('up', 'down');
        if (delta > 0) el.classList.add('up');
        if (delta < 0) el.classList.add('down');
    }

    function setPeriod(period) {
        currentPeriod = period;
        
        // Update Tabs UI
        document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('active'));
        document.getElementById(`tab-${period}`).classList.add('active');

        const now = new Date();
        const todayStart = new Date(now.getFullYear(), now.getMonth(), now.getDate());
        const { start, end, granularity } = buildPeriodRange(period, todayStart);

        // Update Label
        document.getElementById('current-range-label').innerText = period === 'today' ? toLocalDateString(start) : `${toLocalDateString(start)} ~ ${toLocalDateString(end)}`;

        const currentData = allData.filter(d => d.dateObj >= start && d.dateObj <= end);
        currentFilteredData = currentData;

        renderMetrics(allData, todayStart);
        renderCharts(currentData, granularity, start, end);
        renderList(currentData);
    }

    function renderMetrics(data, todayStart) {
        const fmt = (v) => '¥' + v.toLocaleString('zh-CN', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
        const periodMap = {
            'today': { label: '当日', prevLabel: '较昨天', range: buildPeriodRange('today', todayStart) },
            'week': { label: '本周', prevLabel: '较上周', range: buildPeriodRange('week', todayStart) },
            'month': { label: '本月', prevLabel: '较上月', range: buildPeriodRange('month', todayStart) },
            'year': { label: '本年', prevLabel: '较去年', range: buildPeriodRange('year', todayStart) }
        };
        
        const period = periodMap[currentPeriod];
        if (!period) return;
        
        const curr = expenseStats(data, period.range.start, period.range.end);
        const prev = expenseStats(data, period.range.prevStart, period.range.prevEnd);
        
        // 计算平均值和最大值
        const expenseItems = data.filter(d => d.dateObj >= period.range.start && d.dateObj <= period.range.end && d.Category !== '收入' && d.Amount > 0);
        const average = curr.count > 0 ? curr.amount / curr.count : 0;
        const max = expenseItems.length > 0 ? Math.max(...expenseItems.map(item => item.Amount)) : 0;
        
        // 更新UI
        document.getElementById('metric-label-amount').innerText = `${period.label}支出`;
        document.getElementById('val-expense-current').innerText = fmt(curr.amount);
        document.getElementById('val-count-current').innerText = String(curr.count);
        document.getElementById('val-average-current').innerText = fmt(average);
        document.getElementById('val-max-current').innerText = fmt(max);
        
        applyChangeEl('val-expense-change', period.prevLabel, curr.amount, prev.amount);
        applyChangeEl('val-count-change', period.prevLabel, curr.count, prev.count);
        
        // 计算平均值变化
        const prevAverage = prev.count > 0 ? prev.amount / prev.count : 0;
        applyChangeEl('val-average-change', period.prevLabel, average, prevAverage);
    }

    function renderCharts(data, granularity, start, end) {

        // 2. Trend Chart
        const trendMap = {};
        
        // Initialize Trend Map with 0s for complete timeline
        if (granularity === 'hour') {
            for(let i=0; i<24; i++) trendMap[i] = 0;
            expenseData.forEach(d => {
                const h = d.dateObj.getHours();
                trendMap[h] = (trendMap[h] || 0) + d.Amount;
            });
        } else if (granularity === 'day') {
            for (let d = new Date(start); d <= end; d.setDate(d.getDate() + 1)) {
                const k = toLocalDateString(d);
                trendMap[k] = 0;
            }
            expenseData.forEach(d => {
                const k = toLocalDateString(d.dateObj);
                trendMap[k] = (trendMap[k] || 0) + d.Amount;
            });
        } else if (granularity === 'month') {
            for(let i=0; i<12; i++) trendMap[i] = 0;
            expenseData.forEach(d => {
                const m = d.dateObj.getMonth();
                trendMap[m] = (trendMap[m] || 0) + d.Amount;
            });
        }

        const xData = Object.keys(trendMap);
        const yData = Object.values(trendMap);
        const xLabels = xData.map(k => {
            if (granularity === 'hour') return `${k}:00`;
            if (granularity === 'month') return `${parseInt(k)+1}月`;
            if (granularity === 'day') return k.slice(5);
            return k;
        });

        initChart('chart-trend', {
            tooltip: { 
                trigger: 'axis',
                backgroundColor: isDark ? '#1c1c1e' : '#ffffff',
                borderColor: isDark ? '#38383a' : '#e5e7eb',
                textStyle: { color: isDark ? '#ffffff' : '#1c1c1e' }
            },
            grid: { left: '4%', right: '4%', bottom: '10%', top: '10%', containLabel: true },
            xAxis: { 
                type: 'category', 
                data: xLabels,
                axisLine: { show: false },
                axisTick: { show: false },
                axisLabel: { color: textColor, fontSize: 10 }
            },
            yAxis: { 
                type: 'value',
                splitLine: { lineStyle: { color: splitLineColor, type: 'dashed' } },
                axisLabel: { color: textColor, fontSize: 10 }
            },
            series: [{
                data: yData,
                type: 'line',
                smooth: true,
                showSymbol: false,
                lineStyle: { width: 3, color: '#007aff' },
                areaStyle: {
                    color: new echarts.graphic.LinearGradient(0, 0, 0, 1, [
                        { offset: 0, color: 'rgba(0, 122, 255, 0.2)' },
                        { offset: 1, color: 'rgba(0, 122, 255, 0)' }
                    ])
                },
                itemStyle: { color: '#007aff' }
            }]
        });
    }

    function renderList(data) {
        const listEl = document.getElementById('transaction-list');
        listEl.innerHTML = '';

        const recent = [...data].sort((a, b) => b.dateObj - a.dateObj).slice(0, 100);
        let sorted = [...recent];
        if (currentListSort === 'amount-asc') {
            sorted.sort((a, b) => a.Amount - b.Amount);
        } else if (currentListSort === 'amount-desc') {
            sorted.sort((a, b) => b.Amount - a.Amount);
        } else if (currentListSort === 'time-desc') {
            sorted.sort((a, b) => b.dateObj - a.dateObj);
        }
        if (sorted.length === 0) {
            listEl.innerHTML = `
                <li class="empty-state">
                    <div style="font-size: 48px; margin-bottom: 16px;">📊</div>
                    <div style="font-size: 16px; font-weight: 600; margin-bottom: 8px;">本时段无交易记录</div>
                    <div style="font-size: 14px; color: var(--text-sec);">
                        ${currentPeriod === 'today' ? '今天还没有消费记录' : 
                          currentPeriod === 'week' ? '本周还没有消费记录' : 
                          currentPeriod === 'month' ? '本月还没有消费记录' : '本年还没有消费记录'}
                    </div>
                    <div style="margin-top: 16px; font-size: 12px; color: var(--text-sec);">
                        提示：尝试切换到其他时间段查看历史数据
                    </div>
                </li>
            `;
            return;
        }

        sorted.forEach(item => {
            const isIncome = item.Category === '收入' || item.Amount < 0;
            const emoji = categoryEmojis[item.Category] || "✨";
            const amountClass = isIncome ? 'amount-income' : 'amount-expense';
            const amountPrefix = isIncome ? '+' : '-';
            
            const dateStr = item.dateObj.toLocaleDateString('zh-CN', { month: '2-digit', day: '2-digit' });
            const timeStr = item.dateObj.toLocaleTimeString('zh-CN', { hour: '2-digit', minute: '2-digit' });
            
            const metaParts = [dateStr, timeStr];
            if (item.Account) metaParts.push(item.Account);
            if (item.Item || item.Merchant) metaParts.push(item.Item || item.Merchant);
            
            const metaStr = metaParts.join(' · ');
            const notesHtml = item.Notes ? `<div class="tx-notes">${item.Notes}</div>` : '';

            const li = document.createElement('li');
            li.className = 'transaction-item';
            li.innerHTML = `
                <div class="tx-left">
                    <div class="tx-icon">${emoji}</div>
                    <div class="tx-info">
                        <span class="tx-category">${item.Category || '未分类'}</span>
                        <span class="tx-meta">${metaStr}</span>
                        ${notesHtml}
                    </div>
                </div>
                <div class="tx-right">
                    <span class="tx-amount ${amountClass}">${amountPrefix}¥${Math.abs(item.Amount).toFixed(2)}</span>
                </div>
            `;
            listEl.appendChild(li);
        });
    }

    function initChart(domId, option) {
        if (typeof echarts === 'undefined') {
            console.warn('ECharts not loaded');
            return;
        }
        const dom = document.getElementById(domId);
        if (!dom) return;
        
        let chart = charts[domId];
        if (!chart) {
            chart = echarts.init(dom);
            charts[domId] = chart;
        }
        chart.setOption(option);
    }
