<script>
    // Placeholder to be replaced by Shortcut
    const RAW_DATA = `JSON_DATA_PLACEHOLDER`;

    const CATEGORIES = [
        "Dining", "Transport", "Shopping", "Home", "Entertainment",
        "Social", "Health", "Education", "Finance", "Parents", "Other"
    ];

    // Initialize
    document.addEventListener('DOMContentLoaded', () => {
        setupCategories();
        prefillData();
    });

    function setupCategories() {
        const container = document.getElementById('category-chips');
        const hiddenInput = document.getElementById('category');

        CATEGORIES.forEach(cat => {
            const chip = document.createElement('div');
            chip.className = 'chip';
            chip.textContent = cat;
            chip.onclick = () => selectCategory(cat);
            container.appendChild(chip);
        });
    }

    function selectCategory(cat) {
        const chips = document.querySelectorAll('.chip');
        const hiddenInput = document.getElementById('category');

        chips.forEach(c => {
            if (c.textContent === cat) {
                c.classList.add('selected');
            } else {
                c.classList.remove('selected');
            }
        });

        hiddenInput.value = cat;
    }

    function prefillData() {
        let data = {};
        
        // Parse JSON if valid, otherwise use defaults
        try {
            // Check if placeholder was replaced (not equal to the literal string)
            // Or if it is the literal string, we might want to use empty defaults for testing
            if (RAW_DATA !== 'JSON_DATA_PLACEHOLDER') {
                data = JSON.parse(RAW_DATA);
            }
        } catch (e) {
            console.error("Invalid JSON data", e);
        }

        const now = new Date();
        
        // Set Date
        const dateInput = document.getElementById('date');
        // If data.date is provided (YYYY-MM-DD), use it. Else today.
        dateInput.value = data.date || now.toISOString().split('T')[0];

        // Set Time
        const timeInput = document.getElementById('time');
        // If data.time provided (HH:MM), use it. Else now.
        const hours = String(now.getHours()).padStart(2, '0');
        const minutes = String(now.getMinutes()).padStart(2, '0');
        timeInput.value = data.time || `${hours}:${minutes}`;

        // Set Amount
        if (data.amount) {
            document.getElementById('amount').value = data.amount;
        }

        // Set Category
        if (data.category && CATEGORIES.includes(data.category)) {
            selectCategory(data.category);
        } else {
            // Default to 'Other' or first item if needed, or leave empty
            // selectCategory('Other');
        }

        // Set Item & Merchant
        if (data.item) document.getElementById('item').value = data.item;
        if (data.merchant) document.getElementById('merchant').value = data.merchant;
    }

    function handleCancel() {
        // Shortcuts logic: output nothing or specific cancel string?
        // Usually, closing the web view without output is enough for "Cancel" flow if handled in Shortcuts
        // Or we can output a specific flag.
        // For now, let's just clear body to indicate exit or use window.close()
        // window.close() might not work in all web views.
        document.body.innerHTML = ""; 
    }

    function handleSubmit(e) {
        e.preventDefault();

        const date = document.getElementById('date').value;
        const time = document.getElementById('time').value;
        const amount = document.getElementById('amount').value;
        const category = document.getElementById('category').value;
        const item = document.getElementById('item').value || "";
        const merchant = document.getElementById('merchant').value || "";

        if (!amount || !category) {
            alert("Please fill in Amount and Category.");
            return;
        }

        // CSV Format: Date,Time,Amount,Category,Item,Merchant
        // Ensure no commas in text fields to break CSV, or wrap in quotes if needed.
        // Simple approach: remove commas from text inputs
        const cleanItem = item.replace(/,/g, " ");
        const cleanMerchant = merchant.replace(/,/g, " ");

        const csvLine = `${date},${time},${amount},${category},${cleanItem},${cleanMerchant}`;

        // Replace entire body content with just the CSV string
        // This allows Shortcuts "Get Text from Web View Input" to capture exactly this string.
        document.body.innerText = csvLine;
    }
</script>
