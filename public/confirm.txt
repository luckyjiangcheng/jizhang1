<script>
    const rawData = `JSON_DATA_PLACEHOLDER`;

    function endWith(value) {
        if (window.completion) {
            window.completion(value);
            return;
        }
        document.body.innerText = value;
    }

    function parseData() {
        if (rawData === "JSON_DATA_PLACEHOLDER") return {};
        try {
            return JSON.parse(rawData);
        } catch (_) {
            return {};
        }
    }

    function normalizeDate(dateText) {
        if (dateText) return String(dateText);
        const now = new Date();
        const yyyy = now.getFullYear();
        const mm = String(now.getMonth() + 1).padStart(2, "0");
        const dd = String(now.getDate()).padStart(2, "0");
        return `${yyyy}-${mm}-${dd}`;
    }

    function normalizeTime(timeText) {
        if (timeText) return String(timeText);
        return "";
    }

    function normalizeAmount(amount) {
        if (amount === undefined || amount === null || amount === "") return "";
        const n = Number(amount);
        if (Number.isNaN(n)) return "";
        return n.toFixed(2);
    }

    function cleanText(value) {
        return String(value || "").replace(/,/g, " ");
    }

    function toCsvLine(data) {
        const date = normalizeDate(data.date);
        const time = normalizeTime(data.time);
        const amount = normalizeAmount(data.amount);
        const category = cleanText(data.category || "其他");
        const item = cleanText(data.item);
        const merchant = cleanText(data.merchant);
        return `${date},${time},${amount},${category},${item},${merchant}`;
    }

    const data = parseData();
    endWith(toCsvLine(data));
</script>
