<script>
    // Placeholder to be replaced by Shortcut
    let RAW_DATA = `JSON_DATA_PLACEHOLDER`;

    const CATEGORIES = [
        "餐饮美食", "交通出行", "购物消费", "居家生活", "休闲娱乐",
        "人情往来", "医疗健康", "教育培训", "金融保险", "孝敬父母", "其他"
    ];

    // Initialize
    document.addEventListener('DOMContentLoaded', () => {
        setupCategories();
        // If placeholder is not replaced, use empty object
        if (RAW_DATA === 'JSON_DATA_PLACEHOLDER') {
             RAW_DATA = '{}';
        }
        prefillData();
    });

    function setupCategories() {
        const container = document.getElementById('category-chips');
        // Clear existing
        container.innerHTML = '';
        
        CATEGORIES.forEach(cat => {
            const chip = document.createElement('div');
            chip.className = 'chip';
            chip.textContent = cat;
            chip.onclick = () => selectCategory(cat);
            container.appendChild(chip);
        });
    }

    function selectCategory(cat) {
        const chips = document.querySelectorAll('.chip');
        const hiddenInput = document.getElementById('category');

        chips.forEach(c => {
            if (c.textContent === cat) {
                c.classList.add('selected');
            } else {
                c.classList.remove('selected');
            }
        });
        
        // Update hidden input
        if(hiddenInput) hiddenInput.value = cat;
    }

    function prefillData() {
        let data = {};
        
        try {
            data = JSON.parse(RAW_DATA);
        } catch (e) {
            console.error("Invalid JSON data", e);
        }

        const now = new Date();
        
        // Set Date
        const dateInput = document.getElementById('date');
        // Use data.date or today
        if (dateInput) dateInput.value = data.date || now.toISOString().split('T')[0];

        // Set Time
        const timeInput = document.getElementById('time');
        // Use data.time or now
        const hours = String(now.getHours()).padStart(2, '0');
        const minutes = String(now.getMinutes()).padStart(2, '0');
        if (timeInput) timeInput.value = data.time || `${hours}:${minutes}`;

        // Set Amount
        if (data.amount) {
            const amtInput = document.getElementById('amount');
            if(amtInput) amtInput.value = data.amount;
        }

        // Set Category
        const catInput = document.getElementById('category');
        if (data.category && CATEGORIES.includes(data.category)) {
            selectCategory(data.category);
            if(catInput) catInput.value = data.category;
        }

        // Set Item & Merchant
        if (data.item) {
             const itemInput = document.getElementById('item');
             if(itemInput) itemInput.value = data.item;
        }
        if (data.merchant) {
             const merchInput = document.getElementById('merchant');
             if(merchInput) merchInput.value = data.merchant;
        }
    }

    function handleCancel() {
        // Clear body to indicate cancellation
        document.body.innerText = "";
    }

    function handleSubmit(e) {
        e.preventDefault(); // Prevent actual form submission

        const date = document.getElementById('date').value;
        const time = document.getElementById('time').value;
        const amount = document.getElementById('amount').value;
        const category = document.getElementById('category').value; // hidden input
        const item = document.getElementById('item').value || "";
        const merchant = document.getElementById('merchant').value || "";

        if (!amount || !category) {
            alert("请填写金额和分类。");
            return;
        }

        // CSV Format: Date,Time,Amount,Category,Item,Merchant
        // Remove commas to prevent CSV breakage
        const cleanItem = item.replace(/,/g, " ");
        const cleanMerchant = merchant.replace(/,/g, " ");

        const csvLine = `${date},${time},${amount},${category},${cleanItem},${cleanMerchant}`;

        // Replace body content with CSV string for Shortcuts to capture
        document.body.innerText = csvLine;

        // Call Shortcuts completion handler to close the web view automatically
        // This is the standard way to return data and close the view in iOS Shortcuts
        if (window.completion) {
            window.completion(csvLine);
        } else {
            // Fallback: If completion handler is not available, just show the text.
            // But ideally, the user should tap "Done" in the top right if auto-close fails.
            // Or we can try window.close()
            window.close();
        }
    }
</script>
